{# Inline SVG icon set. Icons are defined once in a hidden sheet and referenced with , so repeating one in a table costs a reference rather than a copy of the path. All are 16x16 on a 24-unit grid, stroke-based, and inherit currentColor. #} {# Render a status pill: an icon, a colour and a label. Args: state_label: text shown in the pill. tone: one of ok / warning / serious / critical / info / neutral. #} {% macro pill(state_label, tone='neutral') -%} {%- set icon = { 'ok': 'icon-ok', 'warning': 'icon-spinner', 'serious': 'icon-warning', 'critical': 'icon-critical', 'info': 'icon-shield', }.get(tone, 'icon-info') -%} {{ state_label }} {%- endmacro %} {# Map a ClusterState to a pill tone. Keeps the colour decision in one place rather than repeated across the cluster, dashboard and analytics pages. #} {% macro state_pill(state) -%} {%- set tone = { 'for_deletion': 'critical', 'deleting': 'warning', 'management': 'info', 'protected': 'info', 'in_grace': 'info', 'active': 'ok', 'no_target': 'serious', }.get(state.value, 'neutral') -%} {{ pill(state.label, tone) }} {%- endmacro %} {# A standalone icon. #} {% macro icon(name, cls='') -%} {%- endmacro %}
